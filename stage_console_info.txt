3. The "Split Fade" Engine Structure
Here is how your Python classes would look to handle the Delays and Split Times while keeping that 512-channel grid alive:

Python
import numpy as np

class DmxEngine:
    def __init__(self):
        self.current = np.zeros(512)
        self.start = np.zeros(512)
        self.target = np.zeros(512)
        
        # Timing settings for the active transition
        self.delay_in = 0.0
        self.fade_in = 1.0
        self.delay_out = 0.0
        self.fade_out = 1.0
        self.start_time = 0.0

    def calculate_frame(self, now):
        elapsed = now - self.start_time
        
        # Calculate individual progress with delays
        p_in = np.clip((elapsed - self.delay_in) / self.fade_in, 0, 1) if self.fade_in > 0 else 1.0
        p_out = np.clip((elapsed - self.delay_out) / self.fade_out, 0, 1) if self.fade_out > 0 else 1.0
        
        # Determine direction: True for Up-fade, False for Down-fade
        up_mask = self.target > self.start
        
        # Apply split-progress
        progress_array = np.where(up_mask, p_in, p_out)
        self.current = self.start + (self.target - self.start) * progress_array
        
        return self.current.astype(np.uint8)

Python
import numpy as np
import struct
import time

class DmxEngine:
    def __init__(self):
        self.current = np.zeros(512, dtype=np.float32)
        self.start_vals = np.zeros(512, dtype=np.float32)
        self.target_vals = np.zeros(512, dtype=np.float32)
        
        self.fade_in = 1.0
        self.fade_out = 1.0
        self.delay_in = 0.0
        self.delay_out = 0.0
        self.start_time = 0.0
        self.is_running = False

    def trigger_scene(self, target_array, fin, fout, din, dout):
        self.start_vals = np.copy(self.current)
        self.target_vals = np.array(target_array, dtype=np.float32)
        self.fade_in, self.fade_out = fin, fout
        self.delay_in, self.delay_out = din, dout
        self.start_time = time.monotonic()
        self.is_running = True

    def calculate(self):
        if not self.is_running:
            return self.current.astype(np.uint8)
        
        elapsed = time.monotonic() - self.start_time
        
        # Calculate Progress
        p_in = np.clip((elapsed - self.delay_in) / self.fade_in, 0, 1) if self.fade_in > 0 else 1.0
        p_out = np.clip((elapsed - self.delay_out) / self.fade_out, 0, 1) if self.fade_out > 0 else 1.0
        
        # Mask: True for channels going UP
        up_mask = self.target_vals > self.start_vals
        prog_array = np.where(up_mask, p_in, p_out)
        
        self.current = self.start_vals + (self.target_vals - self.start_vals) * prog_array
        
        if p_in >= 1.0 and p_out >= 1.0:
            self.is_running = False
            
        return self.current.astype(np.uint8)

    def get_enttec_packet(self, data):
        # Enttec Pro Protocol: Label 6 is "Output Only Send DMX Packet"
        # 0x7E (Start), 0x06 (Label), Length LSB, Length MSB, 0x00 (Start Code), Data..., 0xE7 (End)
        header = struct.pack('<BBH', 0x7E, 0x06, len(data) + 1)
        return header + b'\x00' + data.tobytes() + b'\xE7'
